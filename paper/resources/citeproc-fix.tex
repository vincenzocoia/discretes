% Pandoc 3 citeproc + JOSS template: define commands expected in body and refs.
\providecommand{\citeproc}[2]{\hyperlink{cite.#1}{#2}}
\providecommand{\citeproctext}{}
% \bibitem prints its (empty) optional label as "[]" via \@biblabel; pandoc's
% own default template suppresses this. The vendored JOSS template does not, so
% do it here to avoid an empty bracket "marker" before every reference.
\makeatletter
\def\@biblabel#1{}
\makeatother
% Modern pandoc citeproc emits paragraph-style reference entries (no \item),
% but the vendored JOSS template defines CSLReferences as a list environment.
% Redefine it to paragraph form so the bibliography compiles.
\renewenvironment{CSLReferences}[2]
 {\setlength{\parindent}{0pt}%
  \everypar{\setlength{\hangindent}{\ifodd #1 \cslhangindent\else 0pt\fi}}%
  \ignorespaces}%
 {\par}
